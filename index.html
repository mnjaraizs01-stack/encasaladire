    function nextStep(currentStep) {

        document.getElementById(`step-${currentStep}`).classList.remove('active');

        document.getElementById(`step-${currentStep + 1}`).classList.add('active');

    }


    function validateAndNext(currentStep, inputId) {

        const input = document.getElementById(inputId);

        const error = document.getElementById(`err-${inputId}`);

        

        if (!input.value.trim()) {

            error.style.display = 'block';

            input.focus();

            return;

        } else {

            error.style.display = 'none';

        }

        

        nextStep(currentStep);

    }


    document.getElementById('partyForm').addEventListener('submit', function(e) {

        e.preventDefault();

        

        const lastInput = document.getElementById('anfitriona');

        if (!lastInput.value.trim()) {

            document.getElementById('err-anfitriona').style.display = 'block';

            return;

        }


        const submitBtn = document.getElementById('submitBtn');

        submitBtn.innerHTML = 'Enviando... ⏳';

        submitBtn.disabled = true;


        const formData = new FormData(this);


        fetch(this.action, {

            method: 'POST',

            body: formData,

            headers: {

                'Accept': 'application/json'

            }

        }).then(response => {

            document.getElementById('step-7').classList.remove('active');

            document.getElementById('step-final').classList.add('active');

        }).catch(error => {

            document.getElementById('step-7').classList.remove('active');

        });

    });
